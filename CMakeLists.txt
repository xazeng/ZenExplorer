cmake_minimum_required(VERSION 3.24)
project(ZenExplorer VERSION 0.1.0 LANGUAGES CXX RC)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "ZenExplorer requires 64-bit Windows")
endif()
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_compile_options(/utf-8 /W4 /permissive- /EHsc)
add_library(zen_core src/model.cpp src/storage.cpp src/platform.cpp src/location.cpp src/quick_access.cpp)
target_include_directories(zen_core PUBLIC src)
target_link_libraries(zen_core PUBLIC shell32 ole32 oleaut32 oleacc shlwapi uuid propsys comctl32)
# 应用和 Shell 回归复用相同实现，资源与入口仍由各可执行文件持有。
add_library(zen_operations STATIC src/operations.cpp src/task_tracker.cpp)
target_link_libraries(zen_operations PUBLIC zen_core)
add_library(zen_worker STATIC src/worker.cpp)
target_link_libraries(zen_worker PUBLIC zen_operations)
add_library(zen_ui STATIC src/app.cpp src/shell_menu.cpp src/shell_view.cpp src/tabs.cpp
  src/ui_icons.cpp src/task_feedback.cpp src/address_bar.cpp src/quick_access_ui.cpp)
target_link_libraries(zen_ui PUBLIC zen_operations comdlg32 gdiplus dwmapi)

add_executable(ZenExplorer WIN32 src/main.cpp src/app.rc src/app.manifest)
target_link_libraries(ZenExplorer PRIVATE zen_ui zen_worker)
install(TARGETS ZenExplorer RUNTIME DESTINATION .)

enable_testing()
add_executable(zen_tests tests/tests.cpp src/app.manifest)
target_link_libraries(zen_tests PRIVATE zen_core)
add_test(NAME model_and_storage COMMAND zen_tests)

add_executable(zen_navigation_tests tests/navigation_tests.cpp src/app.manifest)
target_link_libraries(zen_navigation_tests PRIVATE zen_core)
add_test(NAME navigation_contracts COMMAND zen_navigation_tests)

add_executable(zen_task_tracker_tests tests/task_tracker_tests.cpp src/app.manifest)
target_link_libraries(zen_task_tracker_tests PRIVATE zen_operations)
add_test(NAME task_tracking COMMAND zen_task_tracker_tests)

add_executable(zen_operations_tests tests/operations_tests.cpp src/app.manifest)
target_link_libraries(zen_operations_tests PRIVATE zen_operations)
add_dependencies(zen_operations_tests ZenExplorer)
add_test(NAME independent_operations COMMAND zen_operations_tests)
set_tests_properties(independent_operations PROPERTIES TIMEOUT 150)

add_executable(zen_shell_startup_tests tests/shell_startup_tests.cpp src/app.rc src/app.manifest)
target_link_libraries(zen_shell_startup_tests PRIVATE zen_ui)
add_test(NAME shell_startup_restore COMMAND zen_shell_startup_tests)
set_tests_properties(shell_startup_restore PROPERTIES TIMEOUT 100)

add_executable(zen_shell_commands_tests tests/shell_commands_tests.cpp src/app.rc src/app.manifest)
target_link_libraries(zen_shell_commands_tests PRIVATE zen_ui zen_worker)
add_test(NAME shell_commands COMMAND zen_shell_commands_tests)
set_tests_properties(shell_commands PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE)

foreach(target ZenExplorer zen_shell_startup_tests zen_shell_commands_tests)
  target_include_directories(${target} PRIVATE assets/app-icon)
endforeach()
