cmake_minimum_required(VERSION 3.24)
project(ZenExplorer VERSION 0.1.0 LANGUAGES CXX RC)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "ZenExplorer requires 64-bit Windows")
endif()
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_compile_options(/utf-8 /W4 /permissive- /EHsc)
add_library(zen_core src/model.cpp src/storage.cpp src/platform.cpp)
target_include_directories(zen_core PUBLIC src)
target_link_libraries(zen_core PUBLIC shell32 ole32 oleaut32 oleacc shlwapi uuid propsys comctl32)
add_executable(ZenExplorer WIN32 src/app.rc src/worker.cpp src/main.cpp src/ui_icons.cpp src/app.cpp src/shell_menu.cpp src/shell_view.cpp src/tabs.cpp src/operations.cpp src/app.manifest)
target_include_directories(ZenExplorer PRIVATE assets/app-icon)
target_link_libraries(ZenExplorer PRIVATE zen_core comdlg32 gdiplus dwmapi)
add_executable(zen_tests tests/tests.cpp src/app.manifest)
target_link_libraries(zen_tests PRIVATE zen_core)
enable_testing()
add_test(NAME model_and_storage COMMAND zen_tests)

add_executable(zen_operations_tests tests/operations_tests.cpp src/operations.cpp src/app.manifest)
target_link_libraries(zen_operations_tests PRIVATE zen_core)
add_dependencies(zen_operations_tests ZenExplorer)
add_test(NAME independent_operations COMMAND zen_operations_tests)
set_tests_properties(independent_operations PROPERTIES TIMEOUT 150)

install(TARGETS ZenExplorer RUNTIME DESTINATION .)

add_executable(zen_shell_startup_tests tests/shell_startup_tests.cpp src/app.cpp src/shell_menu.cpp src/shell_view.cpp src/tabs.cpp src/ui_icons.cpp src/operations.cpp src/app.rc src/app.manifest)
target_include_directories(zen_shell_startup_tests PRIVATE assets/app-icon)
target_link_libraries(zen_shell_startup_tests PRIVATE zen_core comdlg32 gdiplus dwmapi)
add_test(NAME shell_startup_restore COMMAND zen_shell_startup_tests)
set_tests_properties(shell_startup_restore PROPERTIES TIMEOUT 100)

target_compile_definitions(zen_tests PRIVATE ZEN_TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")

add_executable(zen_shell_commands_tests tests/shell_commands_tests.cpp src/worker.cpp src/app.cpp src/shell_menu.cpp src/shell_view.cpp src/tabs.cpp src/ui_icons.cpp src/operations.cpp src/app.rc src/app.manifest)
target_include_directories(zen_shell_commands_tests PRIVATE assets/app-icon)
target_link_libraries(zen_shell_commands_tests PRIVATE zen_core comdlg32 gdiplus dwmapi)
add_test(NAME shell_commands COMMAND zen_shell_commands_tests)
set_tests_properties(shell_commands PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE)
